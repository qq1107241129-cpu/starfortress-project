# TASKS/015.5-ui-visual-polish-foundation.md

## 任务目标

第一轮 UI 美化，统一按钮、面板、文字风格，让主界面、战斗界面、结算界面看起来更像"高级精细像素风 + 轻 UI 质感"的科幻小游戏。

## 背景说明

### 当前问题

- 按钮是默认灰按钮，视觉很丑
- 没有统一的颜色主题
- 没有统一的字号规范
- 面板背景没有统一风格

### 美术方向

- 科幻基地 UI
- 深色半透明面板
- 蓝紫色/青色描边感
- 按钮有轻微发光/描边/层次
- 文字清晰，不拥挤
- 不要默认灰按钮
- 不要过度花哨
- 保持竖屏小游戏轻 UI 质感

    panelBgColor: new Color(20, 20, 40, 230),
    panelBorderColor: new Color(100, 180, 255, 200),

### 2. 新增 UIStyleUtil.ts

提供通用方法：

```typescript
import { Node, Label, Button, UITransform, Color, Graphics } from 'cc';
import { UI_THEME } from './UITheme';

/**
 * 样式化按钮
 * 不强行统一所有按钮尺寸，优先保留原 UITransform 尺寸
 */
export function styleButton(buttonNode: Node, variant: 'primary' | 'danger' | 'secondary' = 'primary'): void {
    const graphics = buttonNode.getComponent(Graphics) || buttonNode.addComponent(Graphics);
    const transform = buttonNode.getComponent(UITransform);
    
    // 优先保留原 UITransform 尺寸
    let width = 200;
    let height = 60;
    if (transform) {
        width = transform.width;
        height = transform.height;
    }
    
    let bgColor: Color;
    let borderColor: Color;
    
    switch (variant) {
        case 'primary':
            bgColor = UI_THEME.primaryButtonColor;
            borderColor = UI_THEME.primaryButtonBorderColor;
            break;
        case 'danger':
            bgColor = UI_THEME.dangerButtonColor;
            borderColor = UI_THEME.dangerButtonBorderColor;
            break;
        case 'secondary':
            bgColor = UI_THEME.secondaryButtonColor;
            borderColor = UI_THEME.secondaryButtonBorderColor;
            break;
    }
    
    // 只在初始化时绘制，不做每帧重绘
    graphics.clear();
    graphics.fillColor = bgColor;
    graphics.roundRect(-width / 2, -height / 2, width, height, UI_THEME.buttonRadius);
    graphics.fill();
    graphics.strokeColor = borderColor;
    graphics.lineWidth = 2;
    graphics.roundRect(-width / 2, -height / 2, width, height, UI_THEME.buttonRadius);
    graphics.stroke();
}

/**
 * 样式化文字
 */
export function styleLabel(labelNode: Node, variant: 'title' | 'large' | 'medium' | 'small' = 'medium'): void {
    const label = labelNode.getComponent(Label);
    if (!label) return;
    
    switch (variant) {
        case 'title':
            label.fontSize = UI_THEME.fontSizeTitle;
            label.color = UI_THEME.textPrimaryColor;
            break;
        case 'large':
            label.fontSize = UI_THEME.fontSizeLarge;
            label.color = UI_THEME.textPrimaryColor;
            break;
        case 'medium':
            label.fontSize = UI_THEME.fontSizeMedium;
            label.color = UI_THEME.textSecondaryColor;
            break;
        case 'small':
            label.fontSize = UI_THEME.fontSizeSmall;
            label.color = UI_THEME.textSecondaryColor;
            break;
    }
}

/**
 * 样式化面板
 * 必须读取 UITransform 尺寸，不要用 panelPadding 画 40×40 小面板
 */
export function stylePanel(panelNode: Node): void {
    const graphics = panelNode.getComponent(Graphics) || panelNode.addComponent(Graphics);
    const transform = panelNode.getComponent(UITransform);
    
    // 读取 UITransform 尺寸
    let width = 400;
    let height = 300;
    if (transform) {
        width = transform.width;
        height = transform.height;
    }
    
    // 只在初始化时绘制，不做每帧重绘
    graphics.clear();
    graphics.fillColor = UI_THEME.panelBgColor;
    graphics.roundRect(-width / 2, -height / 2, width, height, UI_THEME.panelRadius);
    graphics.fill();
    graphics.strokeColor = UI_THEME.panelBorderColor;
    graphics.lineWidth = 2;
    graphics.roundRect(-width / 2, -height / 2, width, height, UI_THEME.panelRadius);
    graphics.stroke();
}
```

## 验收标准

1. 主界面按钮不再是默认灰按钮
2. 战斗 UI 按钮风格统一
3. 肉鸽面板、塔位选择面板、塔详情面板风格统一
4. 结算界面按钮风格统一
5. 所有按钮仍可点击
6. UI 不遮挡核心战斗画面

8. 不修改战斗/数值/技能逻辑

10. Web 预览由用户人工确认
